cmake_minimum_required(VERSION 3.22.1)
project(salag_llm LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_BUILD_TYPE Release CACHE STRING "Optimized inference in every APK" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(flag LLAMA_BUILD_COMMON LLAMA_BUILD_TESTS LLAMA_BUILD_EXAMPLES LLAMA_BUILD_TOOLS LLAMA_BUILD_SERVER LLAMA_BUILD_APP LLAMA_OPENSSL GGML_OPENMP GGML_LLAMAFILE GGML_CPU_KLEIDIAI GGML_NATIVE)
    set(${flag} OFF CACHE BOOL "" FORCE)
endforeach()
set(GGML_CPU_ARM_ARCH "armv8.2-a+dotprod" CACHE STRING "" FORCE)
add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../../../../third_party/llama.cpp build-llama)
add_library(salag_llm SHARED salag_llm.cpp)
target_link_libraries(salag_llm PRIVATE llama ggml android log)
